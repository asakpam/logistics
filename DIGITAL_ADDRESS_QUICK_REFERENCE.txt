━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ VALID EXAMPLES:
   GA-001-1234    (Ghana - 001 - 1234)
   GA-123-4567    (Ghana - 123 - 4567)
   GA-999-9999    (Ghana - 999 - 9999)
   GH-001-1234    (Alternative Ghana code)

❌ INVALID EXAMPLES:
   ga-001-1234    (Lowercase - must be uppercase)
   GA-01-1234     (Wrong digit count - need 3 digits)
   GA001-1234     (Missing dash - must use XX-###-####)
   GAA-001-1234   (Too many letters - need 2 letters)
   GA-001         (Missing digits - need 4 digits)
   01-1234        (Missing country code - need 2 letters)
